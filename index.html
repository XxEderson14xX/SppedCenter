Ve a tu proyecto en Supabase // 2. Menú izquierdo → "Edge Functions" // 3. Clic en "Create a new function" // 4. Nombre: crear-usuario (exactamente así, con guión) // 5. Borra todo el código de ejemplo y pega este archivo completo // 6. Clic en "Deploy" // ============================================================================ import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", }; Deno.serve(async (req) => { // Responder a preflight CORS if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } try { // ── 1. Verificar que quien llama es un administrador activo ────────────── const authHeader = req.headers.get("Authorization"); if (!authHeader) { return respError("No autorizado.", 401); } // Cliente con el token del usuario que hace la petición (para verificar rol) const sbCaller = createClient( Deno.env.get("SUPABASE_URL"), Deno.env.get("SUPABASE_ANON_KEY"), { global: { headers: { Authorization: authHeader } } } ); const { data: { user: caller }, error: errAuth } = await sbCaller.auth.getUser(); if (errAuth || !caller) return respError("No autorizado.", 401); const { data: perfil } = await sbCaller .from("perfiles") .select("rol, activo") .eq("id", caller.id) .single(); if (!perfil || perfil.rol !== "administrador" || !perfil.activo) { return respError("Solo administradores pueden crear usuarios.", 403); } // ── 2. Leer y validar el cuerpo de la petición ─────────────────────────── const body = await req.json(); const { nombre_completo, username, correo, password, rol } = body; if (!nombre_completo?.trim()) return respError("El nombre completo es obligatorio.", 400); if (!username?.trim()) return respError("El nombre de usuario es obligatorio.", 400); if (!correo?.trim()) return respError("El correo es obligatorio.", 400); if (!password?.trim()) return respError("La contraseña es obligatoria.", 400); if (password.length < 6) return respError("La contraseña debe tener al menos 6 caracteres.", 400); const rolesValidos = ["administrador", "recepcion", "consulta"]; if (!rolesValidos.includes(rol)) return respError("Rol inválido.", 400); // Validar formato de correo básico if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(correo)) { return respError("El formato del correo no es válido.", 400); } // Username: solo letras minúsculas, números y guión bajo, mínimo 3 chars const usernameLimpio = username.trim().toLowerCase(); if (!/^[a-z0-9_]{3,30}$/.test(usernameLimpio)) { return respError("El usuario solo puede tener letras minúsculas, números y guión bajo (3-30 caracteres).", 400); } // ── 3. Cliente admin (service_role) para crear en auth.users ───────────── const sbAdmin = createClient( Deno.env.get("SUPABASE_URL"), Deno.env.get("SUPABASE_SERVICE_ROLE_KEY") ); // Verificar que el username no esté ocupado const { data: usernameExiste } = await sbAdmin .from("perfiles") .select("id") .eq("username", usernameLimpio) .maybeSingle(); if (usernameExiste) { return respError(`El nombre de usuario "@${usernameLimpio}" ya está en uso.`, 409); } // ── 4. Crear el usuario en auth.users ──────────────────────────────────── const { data: nuevoUsuario, error: errCreate } = await sbAdmin.auth.admin.createUser({ email: correo.trim().toLowerCase(), password, email_confirm: true, // auto-confirmar, no requiere verificar correo user_metadata: { nombre_completo: nombre_completo.trim() }, }); if (errCreate) { // Supabase devuelve "already registered" si el correo existe if (errCreate.message?.includes("already")) { return respError("Ya existe un usuario con ese correo.", 409); } return respError("Error al crear el usuario: " + errCreate.message, 500); } // ── 5. El trigger handle_new_user ya creó el perfil automáticamente. // Ahora lo actualizamos con username, rol y nombre correcto. ──────── const { error: errPerfil } = await sbAdmin .from("perfiles") .update({ username: usernameLimpio, nombre_completo: nombre_completo.trim(), rol, correo: correo.trim().toLowerCase(), activo: true, }) .eq("id", nuevoUsuario.user.id); if (errPerfil) { // El usuario se creó pero el perfil no se pudo actualizar — no es fatal, // el admin puede editarlo desde el panel de Usuarios. console.warn("Usuario creado pero perfil no actualizado:", errPerfil.message); } return new Response( JSON.stringify({ ok: true, mensaje: `Usuario @${usernameLimpio} creado correctamente.`, id: nuevoUsuario.user.id, }), { headers: { ...corsHeaders, "Content-Type": "application/json" }, status: 201 } ); } catch (err) { console.error("Error inesperado:", err); return respError("Error interno del servidor.", 500); } }); function respError(mensaje, status = 400) { return new Response( JSON.stringify({ ok: false, mensaje }), { headers: { ...corsHeaders, "Content-Type": "application/json" }, status } ); }
